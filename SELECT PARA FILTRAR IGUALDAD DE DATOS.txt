
SELECT 
    e.numeroActa, 
    es.ruc, 
    se.direccion
FROM expediente e
INNER JOIN sede se ON se.idSede = e.idSede
INNER JOIN establecimiento es ON es.idEstablecimiento = se.idEstablecimiento -- Unimos con la tabla establecimiento
WHERE e.numeroActa IN (
    SELECT e.numeroActa
    FROM expediente e
    INNER JOIN sede se ON se.idSede = e.idSede
    GROUP BY e.numeroActa
    HAVING COUNT(e.numeroActa) > 1
);

SELECT 
s.idSede,
s.numeroEstacion, 
(SELECT nombre FROM categoria c WHERE c.idCategoria = s.idCategoria) AS categoria,
e.ruc, e.razonSocial, 
s.nombre, 
s.direccion, 
s.idDistrito, 
s.idProvincia,
(SELECT se.nombre FROM situacion_establecimiento se WHERE se.idSituacionEstablecimiento = s.idSituacionEstablecimiento) AS situacion, 
(SELECT sd.nombre FROM situacion_digemid sd WHERE sd.idSituacionDigemid = s.idSituacionDigemid) AS situacionDigemid, 
(SELECT nombre FROM distrito d WHERE d.idDistrito= s.idDistrito) AS distrito,
(SELECT nombre FROM provincia p WHERE p.idProvincia = s.idProvincia ) AS provincia,
(SELECT COUNT(ex.idExpediente) FROM expediente ex WHERE s.idSede = ex.idSede) AS 'totalInspecciones'
FROM sede s 
INNER JOIN establecimiento e 
ON s.idEstablecimiento = e.idEstablecimiento;
